# list COM port
ifconfig -a

# detailed informations on CAN port
ip -details link show can1 

# Shutdown CAN port
sudo ip link set can1 down

# working NOMINAL 5Mb/s, DATA 5Mb/s
sudo ip link set can1 up type can \
bitrate 5000000 dbitrate 5000000 \
sjw 5 dsjw 5 \
sample-point 0.6875 dsample-point 0.6875 \
restart-ms 1000 fd on

# working NOMINAL 5Mb/s, DATA 10Mb/s
sudo ip link set can1 up type can \
bitrate 5000000 dbitrate 10000000 \
sjw 5 dsjw 2 \
sample-point 0.6875 dsample-point 0.7500 \
restart-ms 1000 fd on

#--------------------------------------------------
# working NOMINAL 1Mb/s, DATA 2Mb/s
sudo ip link set can1 type can bitrate 1000000 dbitrate 2000000 sjw 20 dsjw 10 sample-point 0.7500 dsample-point 0.7500 fd on

# NOT working NOMINAL 1Mb/s, DATA 5Mb/s
sudo ip link set can1 up type can bitrate 1000000 dbitrate 5000000 sjw 20 dsjw 4 sample-point 0.7500 dsample-point 0.7500 fd on

#--------------------------------------------------
# working NOMINAL 4Mb/s, DATA 8Mb/s
sudo ip link set can1 up type can bitrate 4000000 dbitrate 8000000 fd on restart-ms 1000

# send to 0x3F (sub 0) - 32bytes / FD mode / dual data rate - reply to 0x00 (sub 7) - same message
cansend can1 1f8##3.31.32.33.34.35.36.37.38.39.55.75.aa.cc.10.aB.ff.11.00.ff.17.56.78.23.aa.55.00.ff.00.30.32
# send to 0x31 (sub 0) - 24bytes / FD mode / dual data rate - reply to 0x00 (sub 7) - same message
cansend can1 188##3.55.75.aa.cc.10.aB.ff.11.00.ff.17.56.78.23.aa.55.00.ff.00.ff.ff.ff.00.aa
