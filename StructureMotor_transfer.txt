typedef struct __motor_t__
{
    .foc.acq.ia;                 // [A]
    .foc.acq.ib;                 // [A]
    .foc.acq.ic;                 // [A]
    .foc.acq.vBus;               // [V]
    .foc.acq.vExt;               // [V]
    
    .foc.enc.thetaElec;          /*!< Angular electrical position  - [pu] */
    .foc.enc.thetaMech[0];       /*!< Angular mechanical position  - [pu] */
    .foc.enc.thetaIndexRelative; /*!< Index mechanical angle */
    .foc.enc.thetaIndexAbsolute; /*!< First index mechanical angle */
    .foc.enc.thetaAbsolute;      /*!< Absolute angular position - [turns] */
    .foc.enc.turnNb;             /*!< Number of full turns (signed) */
    .foc.enc.thetaDir;           /*!< Rotor direction - -1 = CCW/reverse, 1 = CW/forward */
    .foc.enc.flags;     
    
    .foc.enc.speed.speedHigh;          /*!< High speed estimation -  [rad/s] - diff_angular * scaler = velocity */
    .foc.enc.speed.speedMech[0];       /*!< Mechanical output speed - [rad/s] */
    .foc.enc.speed.speedElec;          /*!< Electrical output speed (pole pairs integration) - [rad/s] */
    .foc.enc.speed.theta[0];           /*!< Angular position of rotor - [pu] */
    
    .foc.cmd.posRef;
    .foc.cmd.velRef;
    .foc.cmd.iqff;
    .foc.cmd.kpCoeff;
    .foc.cmd.kdCoeff;
    .foc.cmd.iSat;
    .foc.cmd.cptTimeout;
    .foc.cmd.index;
    
    .foc.cmd.cmdField.systemEnable;	/*!< Bit 0      : R/W - Enable system to start. No power on motors.         */
    .foc.cmd.cmdField.motorEnable;    	/*!< Bit 1      : R/W - Enable power on motor.                              */
    .foc.cmd.cmdField.rollOverEnable;	/*!< Bit 2      : R/W - Limit the rotations to +/- 128 rounds.              */
    .foc.cmd.cmdField.encOffsetEnable;	/*!< Bit 3      : R/W - Use encoder index in position computation.          */
    
    .foc.cmd.periphField.drv_initErr;	/*!< Bits 0   : R/W - DRV initialization went wrong			    */
    .foc.cmd.periphField.mot_tempErr;	/*!< Bits 1   : R/W - Estimated motor temperature is too high.		    */
    
    .foc.sinTheta;           /*!< Sinus of the electrical rotor angle */
    .foc.cosTheta;           /*!< Cosine of the electrical rotor angle */
    .foc.ialpha;             /*!< Direct Clarke transform result - [A] */
    .foc.ibeta;              /*!< Direct Clarke transform result - [A] */
    .foc.id;                 /*!< Filtered Direct Park transform result - [A] */
    .foc.iq;                 /*!< Filtered Quadrature Park transform result - [A] */
    .foc.ud;                 /*!< Inverse Park transform result - [V] */
    .foc.uq;                 /*!< Inverse Park transform result - [V] */
    .foc.ualpha;             /*!< Inverse Clarke transform result - [V] */
    .foc.ubeta;              /*!< Inverse Clarke transform result - [V] */
    .foc.ua;                 /*!< Phase commands - Channel a - [V] */
    .foc.ub;                 /*!< Phase commands - Channel b - [V] */
    .foc.uc;                 /*!< Phase commands - Channel c - [V] */
    .foc.dtc_u;              /*!< Duty cycles on channel u - [pu] between -0.5 and 0.5 */
    .foc.dtc_v;              /*!< Duty cycles on channel v - [pu] between -0.5 and 0.5 */
    .foc.dtc_w;              /*!< Duty cycles on channel w - [pu] between -0.5 and 0.5 */
    .foc.idRef;              /*!< PI controller set point values for D-axis current - [A] */
    .foc.iqRef;              /*!< PI controller set point values for Q-axis current - [A] */

    .foc.piId.err;		/*!< Current error */
    .foc.piId.integral;           /*!< Integral term partial computation */
    .foc.piId.out;                /*!< Controller output */
    
    .foc.piIq.err;		/*!< Current error */
    .foc.piIq.integral;           /*!< Integral term partial computation */
    .foc.piIq.out;                /*!< Controller output */
    
    .foc.pdPosVel.errTheta;           /*!< Error position */
    .foc.pdPosVel.errSpeed;           /*!< Error speed */
    .foc.pdPosVel.derivative;         /*!< Derivative term partial computation */
    .foc.pdPosVel.out;                /*!< Controller output */    
    
    .itCnt;          /*!< Event counter incrementing on every IT call */
    .id;             /*!< Motor identification */
    .error;          /*!< Bit field for the error sources. */
    .crState;        /*!< Current motor state for the FSM. */
} motor_t;
